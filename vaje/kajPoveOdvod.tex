\textbf{Predstavil: } Klara Glavač Vesel

Dani $f \colon [a,b] \to \mathbb{R}$  in $g \colon [a,b] \to \mathbb{R}$  zvezni  funkciji na $[a,b]$ in odvedljivi na $(a,b)$.
			\begin{enumerate}
				\item Dokažite, da če je $f'(x) = 0$ za vsak $x \in (a,b)$, potem je $f$ konstantna funkcija.
				\item Če velja da $f'(x) = g'(x)$ za vsak $x \in (a,b)$, potem obstaja $k \in \mathbb{R}$, da velja $f(x) = g(x) + k$ za vsak $x \in (a,b)$.
				\item Predpostavimo, da velja $|f(x) - f(y)| \leq (x - y)^n$ za nekaj $n > 1$.  Dokažite, da je $f$ konstantna.
				\item Predpostavimo, da je $f'(x) > g'(x)$ za vsak $x$ in da za nekaj $c \in  (a, b)$  velja $f(c) = g(c).$ Dokažite, da $f(x) > g(x)$ za $x > c$ in $f(x) < g(x)$ za $x < c$.
				\item S primerom pokažite, da prejšnja trditev ne drži brez predpostavke $f(c) = g(c)$.
			\end{enumerate}
