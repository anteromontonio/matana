\textbf{Predstavil:}

Naj bosta $f$ in $g$ funkciji. 
		\begin{enumerate}
			\item Če je $f+g$ odvedljiva v točki $a$, ali sta $f$ in $g$ nujno odvedljivi v $a$? 
			\item Če sta $f \cdot g$ in $f$ odvedljiva v $a$, pod katerimi pogoji za $f$ sledi, da je tudi $g$ odvedljiva v $a$?
			\item Dokažite, da če je $f$ odvedljiva v točki $a$, potem je tudi $|f|$ odvedljiva v $a$, kadar $f(a) \neq 0$. 
			\item Podajte protiprimer prejšnje trditve, če je $f(a) = 0$. 
			\item Dokažite, da če sta $f$ in $g$ odvedljivi v $a$, potem sta tudi funkciji $\max(f,g)$ in $\min(f,g)$, definirani s predpisoma 
			\begin{align*}
				\max(f,g)(x) &= \max\{f(x),g(x)\} \text{ in }\\
				\min(f,g)(x) &= \min\{f(x),g(x)\},
			\end{align*}
			odvedljivi v $a$, kadar $f(a) \neq g(a)$.
			\item Podajte protiprimer prejšnje trditve, če je $f(a) = g(a)$.
			Dobro utemeljite vse odgovore.
		\end{enumerate}
